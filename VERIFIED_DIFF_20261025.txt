+    def _edit_revision(self, event: dict) -> Optional[str]:
+        """Unique id of a ``message_changed`` edit, or None when the event is not an edit.
+
+        The original message keeps its ``ts`` across edits, so a re-sent edit without an edit
+        id hits ``admission_conflict`` at native admission (the payload digest changed while
+        the request key did not). Channel that by deriving one edit id from the edit's own
+        timestamp plus a fingerprint of the updated text, mirroring how Telegram keys an edit
+        by its update id. A redelivered edit keeps the same id and replays the same admitted
+        input; a changed edit is a new input instead of a collision.
+        """
+        updated = event
+        edited = updated.get("edited")
+        if not isinstance(edited, dict):
+            return None
+        edited_ts = edited.get("ts")
+        if not isinstance(edited_ts, str) or not edited_ts:
+            return None
+        text = updated.get("text")
+        if not isinstance(text, str):
+            payload = updated.get("payload")
+            if isinstance(payload, dict):
+                text = payload.get("text")
+            if not isinstance(text, str):
+                text = ""
+        return f"{edited_ts}.{hashlib.sha256(text.encode('utf-8')).hexdigest()[:16]}"

+def test_re_admission_same_request_id_returns_existing_row(tmp_path):
+    # Same (principal, session, request_id) admitted a first time; a re-delivery reaches this
+    # write with a different-enriching payload (merged message event with channel context).
+    # That is an idempotent re-input, not a new write: return the existing row and let the
+    # already-scheduled execution settle it, never refuse with admission_conflict.
+    db = store(tmp_path)
+    import hermes_state_runtime as rt
+    try:
+        epoch = rt.begin_runtime_epoch(db, instance_id='boot')
+        base = dict(epoch=epoch, principal_id='human', session_id='s',
+                    request_id='redeliver')
+        first = rt.admit_session_input(db, **base, payload={'text': 'hello'})
+        assert first['status'] == 'queued'
+        assert rt.admit_session_input(db, **base, payload={'text': 'hello', 'channel_context': 'x'}) == first
+        assert rt.admit_session_input(db, **base, payload={'text': 'hello', 'intent': 'queue'}) == first
+        # A genuinely different request_id is a NEW write (not a re-delivery) and must
+        # create its own admission row; it is never an idempotent re-input.
+        differ = dict(base, request_id='new')
+        new_row = rt.admit_session_input(db, **differ, payload={'text': 'other'})
+        assert new_row['admission_id'] != first['admission_id']
+        assert rt.get_session_admission(db, admission_id=first['admission_id'])['status'] == 'queued'
+        # A genuinely different intent is still a re-delivery on the same request_id, so it
+        # returns the existing row rather than opening a second admission.
+        steer = rt.admit_session_input(db, **base, payload={'text': 'hello'}, intent='steer')
+        assert steer['admission_id'] == first['admission_id']
+        claimed = rt.claim_session_input(db, epoch=epoch, session_id='s')
+        assert claimed['admission_id'] == first['admission_id'] and claimed['status'] == 'started'
+        db.close()
+        db = SessionDB(db_path=tmp_path / 'state.db')
+        new_epoch = rt.begin_runtime_epoch(db, instance_id='replacement')
+        assert new_epoch > epoch
+        assert rt.recover_session_inputs(db, epoch=new_epoch) == 1
+        assert rt.get_session_admission(db, admission_id=first['admission_id'])['status'] == 'unknown'
+        db.close()
+    finally:
+        db.close()
